
import os
import tempfile
import unittest

import app as app_module
import models


class UniversityEventSystemTests(unittest.TestCase):

    def setUp(self):
        handle, self.db_path = tempfile.mkstemp(suffix=".db")
        os.close(handle)

        self.original_database = models.DATABASE
        models.DATABASE = self.db_path

        models.init_db()

        app_module.app.config.update(
            TESTING=True,
            SECRET_KEY="test-secret-key"
        )

        self.client = app_module.app.test_client()

    def tearDown(self):
        models.DATABASE = self.original_database

        if os.path.exists(self.db_path):
            os.remove(self.db_path)

    def register_user(self, email="student@example.com"):
        return self.client.post(
            "/register",
            data={
                "full_name": "Test Student",
                "email": email,
                "password": "Student123!"
            },
            follow_redirects=True
        )

    def login_as(self, email, password):
        return self.client.post(
            "/login",
            data={"email": email, "password": password},
            follow_redirects=True
        )

    def create_test_event(self, capacity=2):
        categories = models.get_categories()

        models.create_event(
            {
                "title": "Test Seminar",
                "description": "A test university seminar",
                "category_id": str(categories[0]["id"]),
                "event_date": "2099-12-31",
                "event_time": "12:00",
                "location": "Main Hall",
                "capacity": str(capacity)
            },
            1
        )

        with models.get_connection() as db:
            return db.execute(
                "SELECT id FROM events WHERE title = ?",
                ("Test Seminar",)
            ).fetchone()["id"]

    def test_home_page_loads(self):
        response = self.client.get("/")
        self.assertEqual(response.status_code, 200)

    def test_user_registration(self):
        response = self.register_user()
        self.assertEqual(response.status_code, 200)

        user = models.get_user_by_email("student@example.com")
        self.assertIsNotNone(user)
        self.assertEqual(user["role"], "user")

    def test_duplicate_email_is_rejected(self):
        self.register_user()
        response = self.register_user()

        self.assertIn(
            "artıq qeydiyyatdan keçib".encode("utf-8"),
            response.data
        )

    def test_login_with_correct_password(self):
        self.register_user()

        response = self.login_as(
            "student@example.com",
            "Student123!"
        )

        self.assertEqual(response.status_code, 200)
        self.assertIn(
            "Sistemə uğurla daxil oldunuz.".encode("utf-8"),
            response.data
        )

    def test_admin_dashboard_requires_admin(self):
        response = self.client.get("/dashboard")
        self.assertEqual(response.status_code, 302)

        self.register_user()
        self.login_as("student@example.com", "Student123!")

        response = self.client.get("/dashboard")
        self.assertEqual(response.status_code, 403)

    def test_user_can_register_for_event(self):
        self.register_user()
        self.login_as("student@example.com", "Student123!")

        event_id = self.create_test_event()

        response = self.client.post(
            f"/events/{event_id}/register",
            follow_redirects=True
        )

        self.assertEqual(response.status_code, 200)

        with models.get_connection() as db:
            count = db.execute(
                "SELECT COUNT(*) AS total FROM registrations"
            ).fetchone()["total"]

        self.assertEqual(count, 1)

    def test_duplicate_event_registration_is_rejected(self):
        self.register_user()
        self.login_as("student@example.com", "Student123!")

        event_id = self.create_test_event()

        self.client.post(f"/events/{event_id}/register")
        response = self.client.post(
            f"/events/{event_id}/register",
            follow_redirects=True
        )

        self.assertIn(
            "artıq qeydiyyatdan keçmisiniz".encode("utf-8"),
            response.data
        )

    def test_event_capacity_is_enforced(self):
        self.register_user("student1@example.com")
        self.login_as("student1@example.com", "Student123!")
        event_id = self.create_test_event(capacity=1)

        self.client.post(f"/events/{event_id}/register")
        self.client.post("/logout")

        self.register_user("student2@example.com")
        self.login_as("student2@example.com", "Student123!")

        response = self.client.post(
            f"/events/{event_id}/register",
            follow_redirects=True
        )

        self.assertIn(
            "boş yer qalmayıb".encode("utf-8"),
            response.data
        )


if __name__ == "__main__":
    unittest.main()
